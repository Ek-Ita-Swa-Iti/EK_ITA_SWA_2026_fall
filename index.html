function lineCopyPlugin(hook, vm) { hook.doneEach(function () { document.querySelectorAll('.markdown-section pre > code').forEach(codeBlock => { if (codeBlock.dataset.rowCopyInitialized) return; codeBlock.dataset.rowCopyInitialized = 'true'; const lines = codeBlock.innerHTML.split('\n'); codeBlock.innerHTML = lines.map((lineHtml, index) => { if (index === lines.length - 1 && lineHtml.trim() === '') return ''; return `
${lineHtml}
`; }).join(''); }); document.querySelectorAll('.line-copy-btn').forEach(btn => { btn.onclick = function (e) { e.stopPropagation(); const contentEl = this.nextElementSibling; // Clone the row so the visible DOM is untouched const clone = contentEl.cloneNode(true); // Remove Prism comment tokens (works across Bash, Python, JS, C#, YAML, etc.) clone.querySelectorAll('.token.comment').forEach(el => el.remove()); let textToCopy = clone.innerText.trimEnd(); // Fallback: If the entire line was just a comment, copy the comment itself if (!textToCopy.trim()) { textToCopy = contentEl.innerText.trim(); } navigator.clipboard.writeText(textToCopy).then(() => { const orig = this.innerText; this.innerText = '✓'; setTimeout(() => { this.innerText = orig; }, 1500); }); }; }); }); }
